    80→function statusBadgeClass(status: string): string {
    81→  switch (status) {
    82→    case 'PUBLISHED':
    83→      return 'border-emerald-500/40 bg-emerald-500/15 text-emerald-300'
    84→    case 'READY':
    85→      return 'border-teal-500/40 bg-teal-500/15 text-teal-300'
    86→    case 'REVIEW':
    87→      return 'border-fuchsia-500/40 bg-fuchsia-500/15 text-fuchsia-300'
    88→    case 'GENERATING':
    89→    case 'RENDERING':
    90→      return 'border-violet-500/40 bg-violet-500/15 text-violet-300 animate-pulse'
    91→    case 'STORY':
    92→    case 'STORYBOARD':
    93→      return 'border-amber-500/40 bg-amber-500/15 text-amber-300'
    94→    default:
    95→      return 'border-zinc-500/40 bg-zinc-500/15 text-zinc-300'
    96→  }

   101→export default function Home() {
   102→  const router = useRouter()
   103→  const { toast } = useToast()
   104→  const [expression, setExpression] = useState('')
   105→  const [starting, setStarting] = useState(false)
   106→  const [mine, setMine] = useState<ContentListItem[]>([])
   107→  const [explore, setExplore] = useState<ExploreItem[]>([])
   108→  const [loadingLists, setLoadingLists] = useState(true)
   109→  const [anonymous, setAnonymous] = useState(true)
   110→
   111→  // 修订对话框（早期阶段的快速版本；成片后的智能修改在创作空间）
   112→  const [revisionFor, setRevisionFor] = useState<ContentListItem | null>(null)
   113→  const [revisionText, setRevisionText] = useState('')
   114→  const [revisionBusy, setRevisionBusy] = useState(false)
   115→
   116→  // 版本时间线对话框
   117→  const [timelineFor, setTimelineFor] = useState<ContentListItem | null>(null)
   118→  const [timeline, setTimeline] = useState<VersionItem[] | null>(null)
   119→  const [timelineBusy, setTimelineBusy] = useState<string | null>(null)
   120→
   121→  // 探索作品播放对话框（视觉内容第一：点开即看）
   122→  const [playingItem, setPlayingItem] = useState<ExploreItem | null>(null)
   123→
   124→  const [busyContentId, setBusyContentId] = useState<string | null>(null)
   125→
   126→  const loadLists = useCallback(async () => {
   127→    setLoadingLists(true)
   128→    try {
   129→      const [mineRes, exploreRes] = await Promise.all([
   130→        fetch('/api/contents?scope=mine'),
   131→        fetch('/api/contents?scope=explore'),
   132→      ])
   133→      const mineJson = (await mineRes.json()) as { items?: ContentListItem[]; anonymous?: boolean }
   134→      const exploreJson = (await exploreRes.json()) as { items?: ExploreItem[] }
   135→      setMine(mineJson.items ?? [])
   136→      setAnonymous(Boolean(mineJson.anonymous))
   137→      setExplore(exploreJson.items ?? [])
   138→    } catch {
   139→      toast({ title: '加载失败', description: '网络似乎打了个盹，稍后再试。', variant: 'destructive' })
   140→    } finally {
   141→      setLoadingLists(false)
   142→    }
   143→  }, [toast])
   144→
   145→  useEffect(() => {
   146→    void loadLists()
   147→  }, [loadLists])
   148→
   149→  const startCreation = useCallback(
   150→    async (text: string) => {
   151→      const trimmed = text.trim()
   152→      if (!trimmed) {
   153→        toast({ title: '还想听你说完', description: '写下你想表达的东西，哪怕只是一个念头。' })
   154→        return
   155→      }
   156→      setStarting(true)
   157→      try {
   158→        const res = await fetch('/api/imagine/start', {
   159→          method: 'POST',
   160→          headers: { 'Content-Type': 'application/json' },
   161→          body: JSON.stringify({ text: trimmed }),
   162→        })
   163→        const json = (await res.json()) as { contentId?: string; createUrl?: string; message?: string; error?: string }
   164→        if (!res.ok) {
   165→          toast({ title: '未能开始', description: (json as { message?: string }).message ?? '请稍后再试', variant: 'destructive' })
   166→          return
   167→        }
   168→        setExpression('')
   169→        // 直达创作空间：想 → 创 的第一步
   170→        if (json.createUrl) {
   171→          router.push(json.createUrl)
   172→          return
   173→        }
   174→        await loadLists()
   175→      } catch {
   176→        toast({ title: '网络错误', description: '请稍后再试。', variant: 'destructive' })
   177→      } finally {
   178→        setStarting(false)
   179→      }
   180→    },
   181→    [loadLists, router, toast],
   182→  )
   183→
   184→  const mutateContent = useCallback(
   185→    async (contentId: string, path: string, body?: unknown, successTitle?: string) => {
   186→      setBusyContentId(contentId)
   187→      try {
   188→        const res = await fetch(`/api/contents/${contentId}${path}`, {
   189→          method: 'POST',
   190→          headers: { 'Content-Type': 'application/json' },
   191→          body: body ? JSON.stringify(body) : undefined,
   192→        })
   193→        const json = (await res.json().catch(() => ({}))) as { message?: string; versionNumber?: number }
   194→        if (!res.ok) {
   195→          toast({ title: '操作未完成', description: json.message ?? '请稍后再试', variant: 'destructive' })
   196→          return false
   197→        }
   198→        if (successTitle) {
   199→          toast({ title: successTitle, description: json.message ?? (json.versionNumber ? `现在是${versionLabel(json.versionNumber)}` : undefined) })
   200→        }
   201→        await loadLists()
   202→        return true
   203→      } catch {
   204→        toast({ title: '网络错误', description: '请稍后再试。', variant: 'destructive' })
   205→        return false
   206→      } finally {
   207→        setBusyContentId(null)
   208→      }
   209→    },
   210→    [loadLists, toast],
   211→  )
   212→
   213→  const openTimeline = useCallback(async (item: ContentListItem) => {
   214→    setTimelineFor(item)
   215→    setTimeline(null)
   216→    try {
   217→      const res = await fetch(`/api/contents/${item.id}/versions`)
   218→      const json = (await res.json()) as { versions?: VersionItem[] }
   219→      setTimeline(json.versions ?? [])
   220→    } catch {
   221→      setTimeline([])
   222→    }
   223→  }, [])
   224→
   225→  const restoreVersion = useCallback(
   226→    async (contentId: string, versionId: string) => {
   227→      setTimelineBusy(versionId)
   228→      try {
   229→        const res = await fetch(`/api/contents/${contentId}/versions/${versionId}/restore`, { method: 'POST' })
   230→        const json = (await res.json().catch(() => ({}))) as { message?: string; versionNumber?: number }
   231→        if (!res.ok) {
   232→          toast({ title: '恢复失败', description: json.message ?? '请稍后再试', variant: 'destructive' })
   233→          return
   234→        }
   235→        toast({ title: `已回到「${versionLabel(json.versionNumber ?? 1)}」`, description: '旧版本仍在时间线上，随时可再回去。' })
   236→        await openTimeline(timelineFor ?? ({ id: contentId } as ContentListItem))
   237→        await loadLists()
   238→      } finally {
   239→        setTimelineBusy(null)
   240→      }
   241→    },
   242→    [loadLists, openTimeline, timelineFor, toast],
   243→  )
   244→
   245→  const submitRevision = useCallback(async () => {
   246→    if (!revisionFor) return
   247→    const text = revisionText.trim()

   249→      toast({ title: COPY.revisionTitle, description: '用一句话描述，比如「再温暖一点」。' })
   250→      return
   251→    }
   252→    setRevisionBusy(true)
   253→    const ok = await mutateContent(revisionFor.id, '/versions', { method: 'revision', changelog: text }, '新一版已开始')
   254→    setRevisionBusy(false)
   255→    if (ok) {
   256→      setRevisionFor(null)
   257→      setRevisionText('')
   258→    }
   259→  }, [mutateContent, revisionFor, revisionText, toast])
   260→
   261→  const remix = useCallback(
   262→    async (item: { id: string; title: string }) => {
   263→      const ok = await mutateContent(item.id, '/remix', undefined, COPY.remixDone)
   264→      if (ok) {
   265→        toast({
   266→          title: `${item.title} · 你的版本`,
   267→          description: '配方已到你手上——去「我的创作」继续创作，画面会为你重新生成。',
   268→        })
   269→      }
   270→    },
   271→    [mutateContent, toast],
   272→  )
   273→
   274→  const exploreCards = useMemo(() => explore.slice(0, 9), [explore])
   275→
   276→  return (

   278→      {/* 顶部导航 */}
   279→      <header className="sticky top-0 z-40 border-b border-white/5 bg-zinc-950/80 backdrop-blur-xl">
   280→        <div className="mx-auto flex h-16 max-w-6xl items-center justify-between px-4 sm:px-6">
   281→          <div className="flex items-center gap-2.5">
   282→            <div className="flex size-9 items-center justify-center rounded-xl bg-gradient-to-br from-violet-500 via-fuchsia-500 to-amber-400 shadow-lg shadow-fuchsia-500/25">
   283→              <Sparkles className="size-5 text-white" aria-hidden />
   284→            </div>
   285→            <div className="leading-tight">
   286→              <div className="text-lg font-semibold tracking-tight">ImagineOS</div>
   287→              <div className="hidden text-[11px] text-zinc-400 sm:block">想到了，就让它发生。</div>
   288→            </div>
   289→          </div>
   290→          <nav className="flex items-center gap-1 text-sm text-zinc-300" aria-label="主导航">
   291→            <a className="rounded-lg px-3 py-2 transition hover:bg-white/5 hover:text-white" href="#imagine">创造</a>
   292→            <a className="rounded-lg px-3 py-2 transition hover:bg-white/5 hover:text-white" href="#mine">我的创作</a>
   293→            <a className="rounded-lg px-3 py-2 transition hover:bg-white/5 hover:text-white" href="#explore">探索</a>
   294→          </nav>
   295→        </div>
   296→      </header>
   297→
   298→      <main className="flex-1">
   299→        {/* 英雄区：第一性问题（评审 P0-2：保持原样） */}
   300→        <section id="imagine" className="relative overflow-hidden px-4 pb-16 pt-14 sm:px-6 sm:pt-20">
   301→          <div aria-hidden className="pointer-events-none absolute inset-0">
   302→            <div className="absolute left-1/2 top-[-20%] h-[420px] w-[720px] -translate-x-1/2 rounded-full bg-violet-600/20 blur-[120px]" />
   303→            <div className="absolute right-[10%] top-[30%] h-[240px] w-[240px] rounded-full bg-fuchsia-500/15 blur-[90px]" />
   304→            <div className="absolute left-[5%] top-[55%] h-[200px] w-[200px] rounded-full bg-amber-400/10 blur-[80px]" />
   305→          </div>
   306→
   307→          <div className="relative mx-auto max-w-3xl text-center">
   308→            <Badge variant="outline" className="mb-6 gap-1.5 border-fuchsia-500/30 bg-fuchsia-500/10 text-fuchsia-300">
   309→              <Wand2 className="size-3" aria-hidden /> AI 原生创作循环 · Alpha
   310→            </Badge>
   311→            <h1 className="text-balance text-4xl font-bold leading-tight tracking-tight sm:text-5xl">
   312→              你想<span className="bg-gradient-to-r from-violet-400 via-fuchsia-400 to-amber-300 bg-clip-text text-transparent">表达</span>
   313→              什么？
   314→            </h1>
   315→            <p className="mx-auto mt-4 max-w-xl text-pretty text-zinc-400">
   316→              不需要会剪辑、会摄影、会写 Prompt。说人话，剩下的交给 AI 创作团队。
   317→            </p>
   318→
   319→            <motion.div
   320→              initial={{ opacity: 0, y: 16 }}
   321→              animate={{ opacity: 1, y: 0 }}
   322→              transition={{ duration: 0.5, ease: 'easeOut' }}
   323→              className="mx-auto mt-8 max-w-2xl"
   324→            >
   325→              <div className="rounded-2xl border border-white/10 bg-zinc-900/70 p-3 shadow-2xl shadow-black/40 backdrop-blur">
   326→                <Textarea
   327→                  value={expression}
   328→                  onChange={(e) => setExpression(e.target.value)}
   329→                  onKeyDown={(e) => {
   330→                    if (e.key === 'Enter' && (e.metaKey || e.ctrlKey)) void startCreation(expression)
   331→                  }}
   332→                  placeholder="我想做一个 45 秒的视频，记录我小时候和父亲一起骑自行车……"
   333→                  className="min-h-[96px] resize-none border-0 bg-transparent px-3 pt-3 text-base text-zinc-100 placeholder:text-zinc-500 focus-visible:ring-0"
   334→                  aria-label="写下你想表达的东西"
   335→                  maxLength={2000}
   336→                />

   344→                  >
   345→                    {starting ? <Loader2 className="size-4 animate-spin" aria-hidden /> : <Rocket className="size-4" aria-hidden />}

   362→
   363→            {/* Creation Loop 主循环 */}

   366→                <span key={s} className="flex items-center gap-2">
   367→                  <span className="flex size-8 items-center justify-center rounded-full border border-white/10 bg-zinc-900 text-[13px] font-medium text-zinc-200">{s}</span>
   368→                  {i < 6 && <ArrowRight className="size-3 text-zinc-600" aria-hidden />}
   369→                </span>
   370→              ))}
   371→            </div>
   372→          </div>
   373→        </section>
   374→
   375→        {/* 我的创作（评审 P1：保持区块，卡片接入创作空间） */}

   378→            icon={<PenLine className="size-4" aria-hidden />}
   379→            title="我的创作"
   380→            desc={anonymous ? '开始一次表达，就会出现在这里。' : '每一个想法都是一个可以继续生长的作品。'}
   381→            right={
   382→              mine.length > 0 ? (
   383→                <Badge variant="secondary" className="bg-zinc-800 text-zinc-300">{mine.length} 个作品</Badge>
   384→              ) : undefined
   385→            }
   386→          />
   387→          {loadingLists ? (
   388→            <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   389→              {[0, 1, 2].map((i) => <SkeletonCard key={i} />)}
   390→            </div>
   391→          ) : mine.length === 0 ? (
   392→            <EmptyState
   393→              icon={<MessageCircleQuestion className="size-8 text-zinc-500" aria-hidden />}
   394→              title={anonymous ? '还没有开始第一次表达' : '这里还空着'}

   398→            <div className="grid gap-4 sm:grid-cols-2 lg:grid-cols-3">
   399→              {mine.map((item) => (
   400→                <ContentCard
   401→                  key={item.id}
   402→                  item={item}
